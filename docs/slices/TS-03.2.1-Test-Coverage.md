```python
# tests/test_security_comprehensive.py
import pytest
from django.test import TestCase, Client
from django.contrib.auth.models import User
from django.core.exceptions import ValidationError
from energy.models import Device, Measurement
from energy.validators.device_validators import DeviceCreateValidator
from energy.views.device_views import device_detail, device_edit, device_delete
from documents.views import upload_document
from energy.mqtt.manager import publish_measurement
import json
import os

class SecurityComprehensiveTest(TestCase):
    """Test completi per la sicurezza del sistema"""
    
    def setUp(self):
        self.client = Client()
        self.user1 = User.objects.create_user('user1', 'user1@test.com', 'password')
        self.user2 = User.objects.create_user('user2', 'user2@test.com', 'password')
        self.device = Device.objects.create(
            name='Test Device',
            device_type='sensor',
            location='Room 1',
            user=self.user1
        )
    
    def test_ownership_verification(self):
        """Test verifica ownership dispositivi"""
        # Test accesso autorizzato
        self.client.force_login(self.user1)
        response = self.client.get(f'/devices/{self.device.id}/')
        self.assertEqual(response.status_code, 200)
        
        # Test accesso non autorizzato
        self.client.force_login(self.user2)
        response = self.client.get(f'/devices/{self.device.id}/')
        self.assertEqual(response.status_code, 404)
    
    def test_csrf_protection(self):
        """Test protezione CSRF"""
        # Test richiesta POST senza token CSRF
        response = self.client.post('/dashboard/update/', {
            'device_id': self.device.id,
            'status': 'active'
        })
        self.assertEqual(response.status_code, 403)
        
        # Test richiesta POST con token CSRF
        csrf_token = self.client.get('/dashboard/').cookies['csrftoken'].value
        response = self.client.post('/dashboard/update/', {
            'device_id': self.device.id,
            'status': 'active',
            'csrfmiddlewaretoken': csrf_token
        })
        self.assertEqual(response.status_code, 200)
    
    def test_sql_injection_prevention(self):
        """Test prevenzione SQL injection"""
        from energy.services.device_service import get_devices_by_type
        
        # Test con input malevolo
        malicious_input = "'; DROP TABLE energy_device; --"
        devices = get_devices_by_type(malicious_input)
        
        # Verificare che la tabella esista ancora
        self.assertTrue(Device.objects.count() > 0)
        # Verificare che non ci siano dispositivi
        self.assertEqual(len(devices), 0)
    
    def test_logging_sanitization(self):
        """Test sanitizzazione logging"""
        device_id = 1
        value = 123.45
        
        publish_measurement(device_id, value)
        
        # Verificare che i log non contengano il valore
        log_content = open('logs/mqtt.log').read()
        self.assertNotIn(str(value), log_content)
        self.assertIn(f'device {device_id}', log_content)
    
    def test_file_upload_validation(self):
        """Test validazione file upload"""
        self.client.force_login(self.user1)
        
        # Test file consentito
        with open('test.pdf', 'rb') as f:
            response = self.client.post('/documents/upload/', {
                'document': f
            })
        self.assertEqual(response.status_code, 200)
        
        # Test file non consentito
        with open('malicious.exe', 'rb') as f:
            response = self.client.post('/documents/upload/', {
                'document': f
            })
        self.assertEqual(response.status_code, 400)
    
    def test_input_validation(self):
        """Test validazione input"""
        # Test dati validi
        valid_data = {
            'name': 'Test Device',
            'device_type': 'sensor',
            'location': 'Room 1'
        }
        validator = DeviceCreateValidator(**valid_data)
        self.assertEqual(validator.name, 'Test Device')
        
        # Test dati non validi
        invalid_data = {
            'name': 'AB',  # Troppo corto
            'device_type': 'invalid',  # Tipo non valido
            'location': 'Room 1'
        }
        with self.assertRaises(ValidationError):
            DeviceCreateValidator(**invalid_data)
    
    def test_rate_limiting(self):
        """Test rate limiting"""
        # Test richieste normali
        for i in range(50):
            response = self.client.get('/api/devices/')
            self.assertEqual(response.status_code, 200)
        
        # Test superamento limite
        for i in range(60):
            response = self.client.get('/api/devices/')
            if i >= 100:
                self.assertEqual(response.status_code, 429)
    
    def test_security_headers(self):
        """Test security headers"""
        response = self.client.get('/')
        
        self.assertEqual(response['X-Frame-Options'], 'DENY')
        self.assertEqual(response['X-Content-Type-Options'], 'nosniff')
        self.assertEqual(response['X-XSS-Protection'], '1; mode=block')
        self.assertIn('Strict-Transport-Security', response)
    
    def test_xss_prevention(self):
        """Test prevenzione XSS"""
        malicious_script = '<script>alert("xss")</script>'
        response = self.client.get(f'/?q={malicious_script}')
        
        # Verificare che il script non sia eseguito
        self.assertNotIn('<script>', response.content.decode())
        self.assertEqual(response['X-XSS-Protection'], '1; mode=block')
    
    def test_clickjacking_protection(self):
        """Test protezione clickjacking"""
        response = self.client.get('/')
        
        # Verificare che l'header X-Frame-Options sia presente
        self.assertEqual(response['X-Frame-Options'], 'DENY')
        
        # Verificare che la pagina non possa essere incorporata
        self.assertIn('X-Frame-Options', response)
```
