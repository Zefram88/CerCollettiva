# energy/domain/entities.py
from dataclasses import dataclass
from datetime import datetime
from typing import Optional, List
from enum import Enum

class DeviceStatus(Enum):
    ACTIVE = "active"
    INACTIVE = "inactive"
    MAINTENANCE = "maintenance"
    ERROR = "error"

class DeviceType(Enum):
    SENSOR = "sensor"
    ACTUATOR = "actuator"
    CONTROLLER = "controller"
    GATEWAY = "gateway"

@dataclass
class Device:
    """Entity rappresentante un dispositivo"""
    id: Optional[int]
    name: str
    device_type: DeviceType
    location: str
    status: DeviceStatus
    description: Optional[str] = None
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None
    
    def __post_init__(self):
        if self.created_at is None:
            self.created_at = datetime.now()
        if self.updated_at is None:
            self.updated_at = datetime.now()
    
    def is_active(self) -> bool:
        """Verifica se il dispositivo è attivo"""
        return self.status == DeviceStatus.ACTIVE
    
    def activate(self) -> None:
        """Attiva il dispositivo"""
        self.status = DeviceStatus.ACTIVE
        self.updated_at = datetime.now()
    
    def deactivate(self) -> None:
        """Disattiva il dispositivo"""
        self.status = DeviceStatus.INACTIVE
        self.updated_at = datetime.now()
    
    def set_maintenance(self) -> None:
        """Imposta il dispositivo in manutenzione"""
        self.status = DeviceStatus.MAINTENANCE
        self.updated_at = datetime.now()
    
    def set_error(self) -> None:
        """Imposta il dispositivo in errore"""
        self.status = DeviceStatus.ERROR
        self.updated_at = datetime.now()

@dataclass
class Measurement:
    """Entity rappresentante una misurazione"""
    id: Optional[int]
    device_id: int
    value: float
    unit: str
    timestamp: datetime
    metadata: Optional[dict] = None
    
    def is_valid(self) -> bool:
        """Verifica se la misurazione è valida"""
        return self.value is not None and self.unit is not None
    
    def to_dict(self) -> dict:
        """Converte la misurazione in dizionario"""
        return {
            'id': self.id,
            'device_id': self.device_id,
            'value': self.value,
            'unit': self.unit,
            'timestamp': self.timestamp.isoformat(),
            'metadata': self.metadata
        }

@dataclass
class User:
    """Entity rappresentante un utente"""
    id: Optional[int]
    username: str
    email: str
    is_active: bool = True
    created_at: Optional[datetime] = None
    
    def __post_init__(self):
        if self.created_at is None:
            self.created_at = datetime.now()
    
    def can_access_device(self, device: Device) -> bool:
        """Verifica se l'utente può accedere al dispositivo"""
        # Logica di business per verificare accesso
        return self.is_active and device.is_active()
```

# energy/domain/repositories.py
from abc import ABC, abstractmethod
from typing import List, Optional
from .entities import Device, Measurement, User

class DeviceRepository(ABC):
    """Repository per la gestione dei dispositivi"""
    
    @abstractmethod
    def save(self, device: Device) -> Device:
        """Salva un dispositivo"""
        pass
    
    @abstractmethod
    def find_by_id(self, device_id: int) -> Optional[Device]:
        """Trova un dispositivo per ID"""
        pass
    
    @abstractmethod
    def find_by_user(self, user_id: int) -> List[Device]:
        """Trova tutti i dispositivi di un utente"""
        pass
    
    @abstractmethod
    def find_by_type(self, device_type: str) -> List[Device]:
        """Trova dispositivi per tipo"""
        pass
    
    @abstractmethod
    def delete(self, device_id: int) -> bool:
        """Elimina un dispositivo"""
        pass
    
    @abstractmethod
    def update(self, device: Device) -> Device:
        """Aggiorna un dispositivo"""
        pass

class MeasurementRepository(ABC):
    """Repository per la gestione delle misurazioni"""
    
    @abstractmethod
    def save(self, measurement: Measurement) -> Measurement:
        """Salva una misurazione"""
        pass
    
    @abstractmethod
    def find_by_device(self, device_id: int, limit: int = 100) -> List[Measurement]:
        """Trova misurazioni per dispositivo"""
        pass
    
    @abstractmethod
    def find_by_date_range(self, device_id: int, start_date: datetime, end_date: datetime) -> List[Measurement]:
        """Trova misurazioni in un range di date"""
        pass
    
    @abstractmethod
    def delete_old_measurements(self, days: int) -> int:
        """Elimina misurazioni vecchie"""
        pass

class UserRepository(ABC):
    """Repository per la gestione degli utenti"""
    
    @abstractmethod
    def find_by_id(self, user_id: int) -> Optional[User]:
        """Trova un utente per ID"""
        pass
    
    @abstractmethod
    def find_by_username(self, username: str) -> Optional[User]:
        """Trova un utente per username"""
        pass
    
    @abstractmethod
    def save(self, user: User) -> User:
        """Salva un utente"""
        pass
```

# energy/use_cases/manage_devices.py
from typing import List, Optional
from energy.domain.entities import Device, DeviceType, DeviceStatus, User
from energy.domain.repositories import DeviceRepository, UserRepository

class ManageDevicesUseCase:
    """Use case per la gestione dei dispositivi"""
    
    def __init__(self, device_repository: DeviceRepository, user_repository: UserRepository):
        self.device_repository = device_repository
        self.user_repository = user_repository
    
    def create_device(self, user_id: int, device_data: dict) -> Device:
        """Crea un nuovo dispositivo"""
        # Verificare che l'utente esista
        user = self.user_repository.find_by_id(user_id)
        if not user:
            raise ValueError("User not found")
        
        # Creare il dispositivo
        device = Device(
            id=None,
            name=device_data['name'],
            device_type=DeviceType(device_data['device_type']),
            location=device_data['location'],
            status=DeviceStatus.ACTIVE,
            description=device_data.get('description')
        )
        
        # Salvare il dispositivo
        return self.device_repository.save(device)
    
    def get_device(self, device_id: int, user_id: int) -> Optional[Device]:
        """Ottiene un dispositivo per ID"""
        device = self.device_repository.find_by_id(device_id)
        if not device:
            return None
        
        # Verificare che l'utente possa accedere al dispositivo
        user = self.user_repository.find_by_id(user_id)
        if not user or not user.can_access_device(device):
            return None
        
        return device
    
    def get_user_devices(self, user_id: int) -> List[Device]:
        """Ottiene tutti i dispositivi di un utente"""
        return self.device_repository.find_by_user(user_id)
    
    def update_device(self, device_id: int, user_id: int, device_data: dict) -> Optional[Device]:
        """Aggiorna un dispositivo"""
        device = self.get_device(device_id, user_id)
        if not device:
            return None
        
        # Aggiornare i campi
        if 'name' in device_data:
            device.name = device_data['name']
        if 'location' in device_data:
            device.location = device_data['location']
        if 'description' in device_data:
            device.description = device_data['description']
        
        return self.device_repository.update(device)
    
    def delete_device(self, device_id: int, user_id: int) -> bool:
        """Elimina un dispositivo"""
        device = self.get_device(device_id, user_id)
        if not device:
            return False
        
        return self.device_repository.delete(device_id)
    
    def activate_device(self, device_id: int, user_id: int) -> Optional[Device]:
        """Attiva un dispositivo"""
        device = self.get_device(device_id, user_id)
        if not device:
            return None
        
        device.activate()
        return self.device_repository.update(device)
    
    def deactivate_device(self, device_id: int, user_id: int) -> Optional[Device]:
        """Disattiva un dispositivo"""
        device = self.get_device(device_id, user_id)
        if not device:
            return None
        
        device.deactivate()
        return self.device_repository.update(device)

from typing import List
from datetime import datetime
from energy.domain.entities import Measurement, Device
from energy.domain.repositories import MeasurementRepository, DeviceRepository

class ManageMeasurementsUseCase:
    """Use case per la gestione delle misurazioni"""
    
    def __init__(self, measurement_repository: MeasurementRepository, device_repository: DeviceRepository):
        self.measurement_repository = measurement_repository
        self.device_repository = device_repository
    
    def create_measurement(self, device_id: int, measurement_data: dict) -> Measurement:
        """Crea una nuova misurazione"""
        # Verificare che il dispositivo esista
        device = self.device_repository.find_by_id(device_id)
        if not device:
            raise ValueError("Device not found")
        
        # Creare la misurazione
        measurement = Measurement(
            id=None,
            device_id=device_id,
            value=measurement_data['value'],
            unit=measurement_data['unit'],
            timestamp=measurement_data.get('timestamp', datetime.now()),
            metadata=measurement_data.get('metadata')
        )
        
        # Verificare che la misurazione sia valida
        if not measurement.is_valid():
            raise ValueError("Invalid measurement data")
        
        return self.measurement_repository.save(measurement)
    
    def get_device_measurements(self, device_id: int, limit: int = 100) -> List[Measurement]:
        """Ottiene le misurazioni di un dispositivo"""
        return self.measurement_repository.find_by_device(device_id, limit)
    
    def get_measurements_by_date_range(self, device_id: int, start_date: datetime, end_date: datetime) -> List[Measurement]:
        """Ottiene le misurazioni in un range di date"""
        return self.measurement_repository.find_by_date_range(device_id, start_date, end_date)
    
    def cleanup_old_measurements(self, days: int = 30) -> int:
        """Pulisce le misurazioni vecchie"""
        return self.measurement_repository.delete_old_measurements(days)
