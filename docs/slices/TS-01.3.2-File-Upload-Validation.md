#### Dopo (Sicuro)
```python
import os
from django.core.exceptions import ValidationError

ALLOWED_EXTENSIONS = ['.pdf', '.png', '.jpg', '.jpeg', '.doc', '.docx', '.txt']
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10MB

def validate_file_upload(file):
    """Valida il file caricato per tipo e dimensione"""
    # Validazione dimensione
    if file.size > MAX_FILE_SIZE:
        raise ValidationError(f"File too large. Maximum size: {MAX_FILE_SIZE / (1024*1024):.1f}MB")
    
    # Validazione estensione
    ext = os.path.splitext(file.name)[1].lower()
    if ext not in ALLOWED_EXTENSIONS:
        raise ValidationError(f"File type not allowed. Allowed types: {', '.join(ALLOWED_EXTENSIONS)}")
    
    # Validazione nome file
    if len(file.name) > 255:
        raise ValidationError("File name too long")
    
    # Validazione caratteri speciali nel nome
    import re
    if not re.match(r'^[a-zA-Z0-9._-]+$', file.name):
        raise ValidationError("File name contains invalid characters")
    
    return True

def upload_document(request):
    if request.method == 'POST':
        try:
            file = request.FILES['document']
            
            # SICURO: Validazione file
            validate_file_upload(file)
            
            # Salvare file
            document = Document(
                name=file.name,
                file=file,
                user=request.user
            )
            document.save()
            
            return JsonResponse({'status': 'success'})
            
        except ValidationError as e:
            return JsonResponse({'status': 'error', 'message': str(e)}, status=400)
        except Exception as e:
            logger.error(f"File upload error: {str(e)}")
            return JsonResponse({'status': 'error', 'message': 'Upload failed'}, status=500)
    
    return render(request, 'documents/upload.html')
```
