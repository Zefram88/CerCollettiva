# energy/validators/device_validators.py
from pydantic import BaseModel, validator, Field
from typing import Optional
import re

class DeviceCreateValidator(BaseModel):
    name: str = Field(..., min_length=3, max_length=100)
    device_type: str = Field(..., min_length=2, max_length=50)
    location: str = Field(..., min_length=2, max_length=200)
    description: Optional[str] = Field(None, max_length=500)
    
    @validator('name')
    def validate_name(cls, v):
        if len(v.strip()) < 3:
            raise ValueError('Name must be at least 3 characters long')
        
        # Prevenire XSS
        if '<' in v or '>' in v or 'script' in v.lower():
            raise ValueError('Name contains invalid characters')
        
        return v.strip()
    
    @validator('device_type')
    def validate_device_type(cls, v):
        allowed_types = ['sensor', 'actuator', 'controller', 'gateway']
        if v.lower() not in allowed_types:
            raise ValueError(f'Invalid device type. Allowed: {", ".join(allowed_types)}')
        return v.lower()
    
    @validator('location')
    def validate_location(cls, v):
        if len(v.strip()) < 2:
            raise ValueError('Location must be at least 2 characters long')
        
        # Prevenire injection
        if any(char in v for char in [';', '--', '/*', '*/']):
            raise ValueError('Location contains invalid characters')
        
        return v.strip()

class DeviceUpdateValidator(BaseModel):

            return v.strip()
        return v

# energy/validators/measurement_validators.py
from pydantic import BaseModel, validator, Field
from datetime import datetime
from typing import Optional

class MeasurementCreateValidator(BaseModel):
    device_id: int = Field(..., gt=0)
    value: float = Field(..., ge=-1000, le=1000)
    unit: str = Field(..., min_length=1, max_length=10)
    timestamp: Optional[datetime] = None
    
    @validator('value')
    def validate_value(cls, v):
        if not isinstance(v, (int, float)):
            raise ValueError('Value must be a number')
        
        if abs(v) > 1000:
            raise ValueError('Value out of range')
        
        return float(v)
    
    @validator('unit')
    def validate_unit(cls, v):
        allowed_units = ['W', 'kW', 'V', 'A', '°C', '%', 'bar', 'm/s']
        if v not in allowed_units:
            raise ValueError(f'Invalid unit. Allowed: {", ".join(allowed_units)}')
        return v

# documents/validators/document_validators.py
from pydantic import BaseModel, validator, Field
from typing import Optional

class DocumentCreateValidator(BaseModel):
    name: str = Field(..., min_length=1, max_length=255)
    description: Optional[str] = Field(None, max_length=1000)
    category: str = Field(..., min_length=2, max_length=50)
    
    @validator('name')
    def validate_name(cls, v):
        if len(v.strip()) < 1:
            raise ValueError('Name cannot be empty')
        
        # Prevenire XSS
        if '<' in v or '>' in v or 'script' in v.lower():
            raise ValueError('Name contains invalid characters')
        
        return v.strip()
    
    @validator('category')
    def validate_category(cls, v):
        allowed_categories = ['manual', 'specification', 'certificate', 'report', 'other']
        if v.lower() not in allowed_categories:
            raise ValueError(f'Invalid category. Allowed: {", ".join(allowed_categories)}')
        return v.lower()

from energy.validators.device_validators import DeviceCreateValidator, DeviceUpdateValidator

def create_device(request):
    if request.method == 'POST':
        try:
            # Validazione input
            validator = DeviceCreateValidator(**request.POST.dict())
            
            # Creazione dispositivo
            device = Device.objects.create(
                name=validator.name,
                device_type=validator.device_type,
                location=validator.location,
                description=validator.description,
                user=request.user
            )
            
            return JsonResponse({'status': 'success', 'device_id': device.id})
            
        except ValidationError as e:
            return JsonResponse({'status': 'error', 'message': str(e)}, status=400)
        except Exception as e:
            logger.error(f"Device creation error: {str(e)}")

    return render(request, 'energy/create_device.html')
