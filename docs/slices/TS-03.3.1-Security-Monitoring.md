security_logger.addHandler(file_handler)

@dataclass
class SecurityEvent:
    """Rappresenta un evento di sicurezza"""
    timestamp: datetime
    event_type: str
    severity: str
    user_id: Optional[int]
    ip_address: str
    user_agent: str
    details: Dict
    resolved: bool = False
    
    def to_dict(self) -> Dict:
        """Converte l'evento in dizionario"""
        return {
            'timestamp': self.timestamp.isoformat(),
            'event_type': self.event_type,
            'severity': self.severity,
            'user_id': self.user_id,
            'ip_address': self.ip_address,
            'user_agent': self.user_agent,
            'details': self.details,
            'resolved': self.resolved
        }

class SecurityMonitor:
    """Monitor per eventi di sicurezza"""
    
    def __init__(self):
        self.events: List[SecurityEvent] = []
        self.alert_thresholds = {
            'failed_login': 5,  # 5 tentativi falliti
            'suspicious_activity': 3,  # 3 attività sospette
            'rate_limit_exceeded': 10,  # 10 superamenti rate limit
        }
        self.alert_cooldown = 300  # 5 minuti
    
    def log_security_event(self, event_type: str, details: Dict, request=None, severity: str = 'INFO') -> None:
        """Registra un evento di sicurezza"""
        # Ottenere informazioni dalla richiesta
        ip_address = self._get_client_ip(request) if request else 'unknown'
        user_agent = request.META.get('HTTP_USER_AGENT', 'unknown') if request else 'unknown'
        user_id = request.user.id if request and request.user.is_authenticated else None
        
        # Creare evento
        event = SecurityEvent(
            timestamp=datetime.now(),
            event_type=event_type,
            severity=severity,
            user_id=user_id,
            ip_address=ip_address,
            user_agent=user_agent,
            details=details
        )
        
        # Aggiungere alla lista
        self.events.append(event)
        
        # Loggare l'evento
        security_logger.info(json.dumps(event.to_dict()))
        
        # Verificare se generare alert
        self._check_alert_conditions(event)
    
    def _get_client_ip(self, request) -> str:
        """Ottiene l'IP del client"""
        x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
        if x_forwarded_for:
            ip = x_forwarded_for.split(',')[0]
        else:
            ip = request.META.get('REMOTE_ADDR')
        return ip
    
    def _check_alert_conditions(self, event: SecurityEvent) -> None:
        """Verifica se l'evento deve generare un alert"""
        # Verificare soglie
        if event.event_type in self.alert_thresholds:
            threshold = self.alert_thresholds[event.event_type]
            
            # Contare eventi recenti dello stesso tipo
            recent_events = self._get_recent_events(event.event_type, event.ip_address)
            
            if len(recent_events) >= threshold:
                self._generate_alert(event, recent_events)
    
    def _get_recent_events(self, event_type: str, ip_address: str, minutes: int = 15) -> List[SecurityEvent]:
        """Ottiene eventi recenti dello stesso tipo"""
        cutoff_time = datetime.now() - timedelta(minutes=minutes)
        
        return [
            event for event in self.events
            if (event.event_type == event_type and
                event.ip_address == ip_address and
                event.timestamp >= cutoff_time)
        ]
    
    def _generate_alert(self, event: SecurityEvent, recent_events: List[SecurityEvent]) -> None:
        """Genera un alert per eventi sospetti"""
        alert_key = f"alert:{event.event_type}:{event.ip_address}"
        
        # Verificare cooldown
        if cache.get(alert_key):
            return
        
        # Creare alert
        alert = {
            'timestamp': datetime.now().isoformat(),
            'event_type': event.event_type,
            'ip_address': event.ip_address,
            'user_id': event.user_id,
            'severity': 'HIGH',
            'message': f"Multiple {event.event_type} events detected",
            'event_count': len(recent_events),
            'details': {
                'recent_events': [e.to_dict() for e in recent_events[-5:]]  # Ultimi 5 eventi
            }
        }
        
        # Loggare alert
        security_logger.warning(json.dumps(alert))
        
        # Impostare cooldown
        cache.set(alert_key, True, self.alert_cooldown)
        
        # Inviare notifica (se configurato)
        self._send_alert_notification(alert)
    
    def _send_alert_notification(self, alert: Dict) -> None:
        """Invia notifica per alert critici"""

        # come email, Slack, PagerDuty, etc.

        return [event.to_dict() for event in self.events[-limit:]]

    """Logga tentativi di login falliti"""

            'reason': 'Invalid credentials'

            'user_id': user.id
        },

from typing import Dict, List, Optional
from datetime import datetime, timedelta
from django.core.cache import cache
from django.conf import settings
import json

class AlertManager:
    """Gestore degli alert di sicurezza"""
    
    def __init__(self):
        self.alert_types = {
            'failed_login': {
                'threshold': 5,
                'window': 15,  # minuti
                'severity': 'HIGH',
                'message': 'Multiple failed login attempts detected'
            },
            'rate_limit_exceeded': {
                'threshold': 10,
                'window': 5,
                'severity': 'MEDIUM',
                'message': 'Rate limit exceeded multiple times'
            },
            'suspicious_activity': {
                'threshold': 3,
                'window': 10,
                'severity': 'HIGH',
                'message': 'Suspicious activity detected'
            }
        }
    
    def check_alert_conditions(self, event_type: str, ip_address: str, user_id: Optional[int] = None) -> bool:
        """Verifica se un evento deve generare un alert"""
        if event_type not in self.alert_types:
            return False
        
        alert_config = self.alert_types[event_type]
        key = f"alert_count:{event_type}:{ip_address}"
        
        # Incrementare contatore
        current_count = cache.get(key, 0) + 1
        cache.set(key, current_count, alert_config['window'] * 60)
        
        # Verificare soglia
        if current_count >= alert_config['threshold']:
            self._create_alert(event_type, ip_address, user_id, alert_config)
            return True
        
        return False
    
    def _create_alert(self, event_type: str, ip_address: str, user_id: Optional[int], config: Dict) -> None:
        """Crea un alert"""
        alert = {
            'id': f"{event_type}_{ip_address}_{datetime.now().timestamp()}",
            'timestamp': datetime.now().isoformat(),
            'event_type': event_type,
            'ip_address': ip_address,
            'user_id': user_id,
            'severity': config['severity'],
            'message': config['message'],
            'resolved': False
        }
        
        # Salvare alert
        self._save_alert(alert)
        
        # Inviare notifica
        self._send_notification(alert)

# monitoring/dashboard.py
from django.shortcuts import render
from django.contrib.auth.decorators import login_required
from django.http import JsonResponse
from django.views.decorators.http import require_http_methods
from .security_monitor import security_monitor
from .alerts import alert_manager
from datetime import datetime, timedelta
import json

@login_required
def security_dashboard(request):
    """Dashboard di monitoraggio sicurezza"""
    # Ottenere eventi recenti
    recent_events = security_monitor.get_security_events(limit=50)
    
    # Ottenere alert attivi
    active_alerts = alert_manager.get_active_alerts()
    
    # Statistiche
    stats = {
        'total_events_today': len([e for e in recent_events if e['timestamp'].startswith(datetime.now().strftime('%Y-%m-%d'))]),
        'failed_logins_today': len([e for e in recent_events if e['event_type'] == 'failed_login']),
        'active_alerts': len(active_alerts),
        'high_severity_events': len([e for e in recent_events if e['severity'] == 'HIGH'])
    }
    
    context = {
        'recent_events': recent_events,
        'active_alerts': active_alerts,
        'stats': stats
    }
    
    return render(request, 'monitoring/security_dashboard.html', context)

@login_required
@require_http_methods(["GET"])
def security_events_api(request):
    """API per ottenere eventi di sicurezza"""
    limit = int(request.GET.get('limit', 100))
    event_type = request.GET.get('type')
    
    events = security_monitor.get_security_events(limit=limit)
    
    if event_type:
        events = [e for e in events if e['event_type'] == event_type]
    
    return JsonResponse({'events': events})

@login_required
@require_http_methods(["POST"])
def resolve_event(request):
    """Risolve un evento di sicurezza"""
    event_id = request.POST.get('event_id')
    
    if security_monitor.resolve_event(event_id):
        return JsonResponse({'status': 'success'})
    else:
        return JsonResponse({'status': 'error'}, status=400)
