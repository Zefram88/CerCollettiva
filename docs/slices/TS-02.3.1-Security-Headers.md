#### Middleware per Security Headers
```python
# core/middleware/security_headers.py
from django.utils.deprecation import MiddlewareMixin
from django.conf import settings

class SecurityHeadersMiddleware(MiddlewareMixin):
    def __init__(self, get_response):
        self.get_response = get_response
        self.security_headers = getattr(settings, 'CUSTOM_SECURITY_HEADERS', {})
    
    def __call__(self, request):
        response = self.get_response(request)
        
        # Aggiungere security headers
        for header, value in self.security_headers.items():
            response[header] = value
        
        # Content Security Policy
        if hasattr(settings, 'CSP_DEFAULT_SRC'):
            csp_policy = self._build_csp_policy()
            response['Content-Security-Policy'] = csp_policy
        
        return response
    
    def _build_csp_policy(self):
        """Costruisce la policy CSP"""
        policy_parts = []
        
        if hasattr(settings, 'CSP_DEFAULT_SRC'):
            policy_parts.append(f"default-src {' '.join(settings.CSP_DEFAULT_SRC)}")
        
        if hasattr(settings, 'CSP_SCRIPT_SRC'):
            policy_parts.append(f"script-src {' '.join(settings.CSP_SCRIPT_SRC)}")
        
        if hasattr(settings, 'CSP_STYLE_SRC'):
            policy_parts.append(f"style-src {' '.join(settings.CSP_STYLE_SRC)}")
        
        if hasattr(settings, 'CSP_IMG_SRC'):
            policy_parts.append(f"img-src {' '.join(settings.CSP_IMG_SRC)}")
        
        if hasattr(settings, 'CSP_FRAME_ANCESTORS'):
            policy_parts.append(f"frame-ancestors {' '.join(settings.CSP_FRAME_ANCESTORS)}")
        
        return '; '.join(policy_parts)
```

### Test 1: Security Headers Presenti
```python
def test_security_headers_present():
    response = client.get('/')
    
    assert response['X-Frame-Options'] == 'DENY'
    assert response['X-Content-Type-Options'] == 'nosniff'
    assert response['X-XSS-Protection'] == '1; mode=block'
    assert 'Strict-Transport-Security' in response
```

### Test 2: Protezione XSS
```python
def test_xss_protection():
    malicious_script = '<script>alert("xss")</script>'
    response = client.get(f'/?q={malicious_script}')
    
    # Verificare che il script non sia eseguito
    assert '<script>' not in response.content.decode()
    assert response['X-XSS-Protection'] == '1; mode=block'
```

### Test 3: Protezione Clickjacking

def test_clickjacking_protection():
